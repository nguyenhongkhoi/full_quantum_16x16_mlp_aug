% TRANG 2: KHỐI 8 QUBIT CUỐI

\begin{tikzpicture}[x=0.82cm, y=-0.8cm]

  \node[font=\large\bfseries, anchor=west] at (-2,-2.4) {Khối 8 qubit cuối của một cây};

  % dây
  \foreach \q in {0,...,7} {
    \node[anchor=east, font=\small] at (-0.3,\q) {$\rho_{\q}$};
    \draw[black!60] (-0.2,\q) -- (24.6,\q);
  }

  % re-upload
  \foreach \q in {0,...,7} { \node[gr] at (0.8,\q) {$R_Y(s_{\q}\bar\alpha_{\q})$}; }

  % lớp 1 (r = 1) và lớp 2 (r = 2) vẽ đầy đủ
  \foreach \L/\r/\x in {1/1/2.4, 2/2/12.0} {
    \foreach \q in {0,...,7} { \node[g1] at (\x,\q) {$U^{(\L)}_{\q}$}; }
    \foreach \i in {0,...,7} {
      \pgfmathtruncatemacro{\j}{mod(\i+\r,8)}
      \pgfmathsetmacro{\gx}{\x+1.0+0.95*\i}
      \draw[red!60!black, thick] (\gx,\i) -- (\gx,\j);
      \node[g2] at (\gx,\i) {};
      \node[g2] at (\gx,\j) {};
    }
    \draw[decorate, decoration={brace, amplitude=5pt}] (\x-0.4,-0.6) -- (\x+8.2,-0.6)
      node[midway, above=5pt, font=\small] {lớp \L\ ($r = \r$)};
  }

  % lớp 3, 4 gộp
  \node[qbox, minimum height=64mm, minimum width=22mm] at (21.6,3.5) {lớp 3\\$r = 3$\\[10pt]lớp 4\\$r = 4$\\[6pt]\scriptsize như lớp 1, 2};

  % đo
  \foreach \q in {0,...,7} { \node[meas] at (24.0,\q) {$\nearrow$}; }
  \node[note, anchor=west, text width=36mm] at (25.0,3.5) {%
    \textbf{Đo:}\\
    $\langle X_i\rangle,\langle Y_i\rangle,\langle Z_i\rangle$: 24\\
    $\langle X_iX_j\rangle,\langle Y_iY_j\rangle,$\\$\langle Z_iZ_j\rangle$, 28 cặp: 84};

  % chú thích dưới
  \node[note, anchor=north west, text width=190mm] at (-2,8.0) {%
    \textcolor{green!45!black}{$R_Y(s_i\bar\alpha_i)$}: re-upload; $\bar\alpha_i$ = góc trung bình của 32 pixel mà qubit $i$ đại diện,
    $s_i$ học được (khởi tạo 0).\\
    Mỗi lớp $\ell$: \textcolor{blue!60!black}{$U^{(\ell)}_i$} = unitary 1 qubit tổng quát (4 tham số) trên từng qubit, rồi
    \textcolor{red!60!black}{$\blacksquare\!-\!\blacksquare$} = unitary 2 qubit tổng quát (16 tham số) trên vòng $(i,\,i{+}r)$,
    lần lượt $i = 0,\dots,7$ (cặp cuối khép vòng).\\
    Tham số: mỗi lớp $8\times4 + 8\times16 = 160$; 4 lớp $= 640$; cộng re-upload 8.};

\end{tikzpicture}
